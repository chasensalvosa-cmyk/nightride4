    46→function categoryIcon(category: ReportCategory): MapMarkerData['icon'] {
    47→  switch (category) {
    48→    case 'road_hazard': return 'hazard'
    49→    case 'damaged_road': return 'damage'
    50→    case 'construction': return 'construction'
    51→    case 'road_closure': return 'closure'
    52→    case 'charging_station': return 'charging'
    53→    case 'riding_location': return 'riding'
    54→    case 'police_activity': return 'police'
    55→    default: return 'safety'
    56→  }
    57→}

   501→  // ------------------------------------------------------------ layer statuses
   502→  const layerStatuses: Partial<Record<keyof typeof layers, LayerStatus>> = useMemo(() => {
   503→    const out: Partial<Record<keyof typeof layers, LayerStatus>> = {

   505→      road_hazards: { count: reports.filter((r) => HAZARD_CATEGORIES.includes(r.category)).length },
   506→      road_closures: { count: reports.filter((r) => CLOSURE_CATEGORIES.includes(r.category)).length },
   507→      riding_areas: { count: reports.filter((r) => RIDING_CATEGORIES.includes(r.category)).length },

   525→
   526→  const basemapProvider = statusQuery.data?.providers.find((p) => p.key === 'mapbox')
   527→  const osmProvider = statusQuery.data?.providers.find((p) => p.key === 'osm')
   528→  const dataProviders = statusQuery.data?.providers.filter((p) =>
   529→    ['aviation', 'charging', 'public_safety'].includes(p.key),

   532→    !!dataProviders && dataProviders.some((p) => p.status === 'connected')

   542→          setSelectedMarkerId(null)
   543→        }}
   544→        onMapError={(msg) => setMapError(msg)}
   545→        userLocation={geo.coords ? { lat: geo.coords.lat, lng: geo.coords.lng } : null}

   556→              void routeHere(r.latitude, r.longitude, r.label)
   557→            }}
   558→          />
   559→          <div className="pointer-events-auto rounded-lg border border-border bg-panel/95 px-3 py-2 backdrop-blur-sm">
   560→            <div className="flex flex-wrap items-center gap-x-3 gap-y-1">
   561→              <span className="flex items-center gap-1.5 text-[10px] font-medium uppercase tracking-wider text-text-muted">
   562→                <StatusDot
   563→                  status={
   564→                    basemapProvider?.status === 'connected' || osmProvider?.status === 'connected'
   565→                      ? 'connected'
   566→                      : 'error'
   567→                  }
   568→                />
   569→                Map
   570→              </span>
   571→              <span className="flex items-center gap-1.5 text-[10px] font-medium uppercase tracking-wider text-text-muted">
   572→                <StatusDot status={dataAnyMissing ? 'error' : dataOk ? 'connected' : 'missing'} />
   573→                Data
   574→              </span>
   575→              <span className="flex items-center gap-1.5 text-[10px] font-medium uppercase tracking-wider text-text-muted">
   576→                <StatusDot status={realtimeConnected ? 'connected' : 'missing'} />
   577→                Live
   578→              </span>
   579→              <span className="ml-auto hidden text-[10px] text-text-muted md:inline">
   580→                {basemapLabel}
   581→              </span>
   582→            </div>
   583→            {mapError && (
   584→              <p className="mt-1 truncate text-[10px] text-status-warning" title={mapError}>
   585→                {mapError}
   586→              </p>
   587→            )}
   588→          </div>
   589→
   590→          {/* fastest-route options (all viewports) */}
   591→          {routeDestination && (
   592→            <RoutePanel
   593→              destination={routeDestination}
   594→              routes={routeOptions}
   595→              activeIndex={activeRouteIdx}
   596→              onSelect={setActiveRouteIdx}
   597→              onClear={() => {
   598→                setRouteDestination(null)
   599→                setActiveRouteIdx(0)
   600→              }}
   601→              fetching={routeQuery.isFetching}
   602→              freshness={routeQuery.data?.freshness ?? 'unavailable'}
   603→              note={routeQuery.data?.note}
   604→              attribution={
   605→                routeQuery.data?.attribution ??
   606→                'Routing © OpenStreetMap contributors, ODbL — OSRM'
   607→              }
   608→            />
   609→          )}
   610→
   611→          {/* location details (desktop) */}
   612→          {selection && (
   613→            <div className="hidden md:block">
   614→              <LocationPanel
   615→                selection={selection}
   616→                onClose={() => {
   617→                  setSelection(null)
   618→                  setSelectedMarkerId(null)
   619→                }}
   620→                onReportHere={openReport}
   621→                onSaveHere={(lat, lng, label) => {
   622→                  setSaveDraft({ lat, lng, label })
   623→                  setSaveOpen(true)
   624→                }}
   625→                onRouteHere={routeHere}
   626→                onConfirmReport={(id) => confirmMutation.mutate(id)}
   627→                onFlagReport={(id) => {
   628→                  setFlagTarget(id)
   629→                  setFlagOpen(true)
   630→                }}
   631→                canModerate={!!user}
   632→              />
   633→            </div>
   634→          )}
   635→        </div>
   636→
   637→        {/* top-right: layers */}
   638→        <div className="absolute right-3 top-3 z-10 w-56 md:w-64">
   639→          <MapLayers
   640→            statuses={layerStatuses}
   641→            onSharePresence={handleSharePresence}
   642→            sharingPresence={sharingPresence}
   643→            canSharePresence={canSharePresence}
   644→          />
   645→        </div>
   646→
   647→        {/* bottom-right: controls + report */}
   648→        <div className="absolute bottom-20 right-3 z-10 flex flex-col items-end gap-2 md:bottom-8">
   649→          <Button
   650→            size="sm"
   651→            onClick={() => openReport(view.lat, view.lng)}
   652→            className="h-9 gap-1.5 rounded-lg"
   653→          >
   654→            <TriangleAlert className="h-3.5 w-3.5" aria-hidden />
   655→            Report
   656→          </Button>
   657→          <MapControls onRecenter={recenter} locating={locating} />
   658→        </div>
   659→
   660→        {/* bottom-left: legend (desktop only) */}
   661→        <div className="absolute bottom-8 left-3 z-10 hidden md:block">
   662→          <MapLegend />
   663→        </div>
   664→
   665→        {/* mobile: location details bottom sheet */}
   666→        {selection && (
   667→          <div className="absolute inset-x-3 bottom-20 z-10 md:hidden">
   668→            <LocationPanel
   669→              selection={selection}
   670→              onClose={() => {
   671→                setSelection(null)
   672→                setSelectedMarkerId(null)
   673→              }}
   674→              onReportHere={openReport}
   675→              onSaveHere={(lat, lng, label) => {
   676→                setSaveDraft({ lat, lng, label })
   677→                setSaveOpen(true)
   678→              }}
   679→              onRouteHere={routeHere}
   680→              onConfirmReport={(id) => confirmMutation.mutate(id)}
   681→              onFlagReport={(id) => {
   682→                setFlagTarget(id)
   683→                setFlagOpen(true)
   684→              }}
   685→              canModerate={!!user}
   686→            />
   687→          </div>
   688→        )}
   689→
   690→        {/* loading indicator while the engine boots */}
   691→        {!mapInstance && (
   692→          <div className="absolute inset-0 z-20 flex items-center justify-center bg-background/80">
   693→            <div className="flex flex-col items-center gap-3">
   694→              <span className="flex h-10 w-10 animate-pulse items-center justify-center rounded-lg border border-border bg-panel">
   695→                <MapPin className="h-5 w-5 text-text-muted" aria-hidden />

   698→            </div>
   699→          </div>
   700→        )}
   701→
   702→        <MarkerRenderer map={mapInstance} markers={markers} />
   703→        {routeDestination && (
   704→          <RouteOverlay
   705→            map={mapInstance}
   706→            routes={routeOptions}
   707→            activeIndex={activeRouteIdx}
   708→            destination={routeDestination}
   709→            fitKey={`${destKey ?? ''}:${activeRouteIdx}`}
   710→          />

   712→      </MapContainer>
   713→
   714→      <ReportSheet open={reportOpen} draft={reportDraft} onOpenChange={setReportOpen} />
   715→      <SaveLocationDialog open={saveOpen} draft={saveDraft} onOpenChange={setSaveOpen} />
   716→      <FlagDialog
   717→        open={flagOpen}
   718→        onOpenChange={setFlagOpen}
   719→        itemType="report"
   720→        itemId={flagTarget}
   721→      />
   722→    </div>
